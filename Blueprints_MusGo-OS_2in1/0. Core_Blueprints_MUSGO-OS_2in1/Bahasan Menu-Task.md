 * Role: Gerbang Ingress untuk Signal dari Notion.
 * Standard: HMAC Validation, Anti-Replay, Non-Blocking.
 */

func HandleNotionWebhook(w http.ResponseWriter, r *http.Request) {
	// 1. Validasi Signature Notion (X-Notion-Signature)
	signature := r.Header.Get("X-Notion-Signature")
	secret := os.Getenv("NOTION_WEBHOOK_SECRET")
	
	body, _ := io.ReadAll(r.Body)
	if !validateHMAC(body, signature, secret) {
		audit.LogSecurityIncident("UNAUTHORIZED_WEBHOOK_ATTEMPT", "Notion", r.RemoteAddr)
		w.WriteHeader(http.StatusUnauthorized)
		return
	}

	// 2. Transformasi ke Event Object MUSGO-OS
	event := eventbus.Event{
		Topic:     "integration.notion.update",
		Payload:   body,
		Timestamp: time.Now().UTC(),
		Nonce:     generateNonce(),
	}

	// 3. Injeksi ke Signed Event Bus
	eventbus.Publish(event)
	
	w.WriteHeader(http.StatusAccepted)
}

func validateHMAC(payload []byte, signature, secret string) bool {
	h := hmac.New(sha256.New, []byte(secret))
	h.Write(payload)
	expectedSignature := hex.EncodeToString(h.Sum(nil))
	return hmac.Equal([]byte(signature), []byte(expectedSignature))
}

	"musgo/framework/persistence"

/*
 * [KERNEL_DIRECTIVE]
 * NOTION_ROLLBACK_MANAGER - Layer 08 (Persistence) & Layer 14 (Billing)
 * Role: Menangani kegagalan transaksi Notion secara atomik.
 * Principle: All-or-Nothing, Zero-Data-Corruption.
 */

type NotionTransaction struct {
	TransactionID string
	UserID        string
	EstimatedCost int64
	StateSnapshot []byte
}

func (n *NotionConnector) ExecuteAtomicSync(ctx context.Context, tx NotionTransaction, syncFunc func() error) error {
	// 1. PHASE: RESERVE (Lock Saldo Kredit)
	if err := billing.ReserveCredit(tx.UserID, tx.EstimatedCost); err != nil {
		return fmt.Errorf("[BILLING_FAILED] Insufficient credits for Notion sync: %v", err)
	}

	// 2. PHASE: EXECUTE (Mencoba Sinkronisasi ke Notion)
	err := syncFunc()

	if err != nil {
		// 3. PHASE: ROLLBACK CASCADE (Jika Gagal)
		audit.LogSecurityIncident("NOTION_SYNC_FAILURE", tx.TransactionID, err.Error())
		
		// Batalkan reservasi kredit secara instan
		billing.CancelReservation(tx.UserID, tx.TransactionID)
		
		// Pulihkan state lokal dari snapshot imutabel
		persistence.RestoreFromSnapshot(tx.StateSnapshot)
		
		return fmt.Errorf("[ROLLBACK_TRIGGERED] Notion API Down, system reverted to stable state: %v", err)
	}

	// 4. PHASE: COMMIT (Jika Berhasil)
	billing.CommitTransaction(tx.UserID, tx.TransactionID)
	audit.LogSuccess("NOTION_SYNC_SUCCESS", tx.TransactionID)
	
	return nil
}

"""
[KERNEL_DIRECTIVE]
N8N_FACTORY_INVOKER - Layer 03 (GoRo Executor) Integration.
Role: Menjalankan Industrial-Grade Chained Workflows.
Standard: mTLS Enforced, Atomic Transaction, Zero-Simulation.
"""
import requests
from musgo.framework.security import vault, signature
from musgo.framework.resilience import atomic_transaction

class N8NFactoryInvoker:
    def __init__(self):
        self.base_url = "https://n8n.internal:5678"
        self.internal_cert = "/etc/musgo/certs/internal_service.pem"

    async def trigger_industrial_workflow(self, workflow_id: str, data: dict):
        """
        Memicu alur kerja n8n setelah mufakat Musyawarah (Layer 04) tercapai.
        """
        # 1. Start Atomic Transaction (Mencegah kehilangan kredit user)
        with atomic_transaction.start() as tx:
            # 2. Persiapan Payload dengan Metadata Kedaulatan
            payload = {
                "workflow_id": workflow_id,
                "input_data": data,
                "metadata": {
                    "trace_id": signature.generate_trace_id(),
                    "agent_id": "executor_n8n_manager_#115",
                    "timestamp_utc": signature.get_utc_now()
                }
            }

            # 3. Eksekusi via mTLS Protected Channel
            try:
                response = requests.post(
                    f"{self.base_url}/api/v1/workflows/{workflow_id}/execute",
                    json=payload,
                    cert=self.internal_cert,
                    timeout=30
                )
                
                if response.status_code == 200:
                    tx.commit() # Transaksi sah
                    return response.json()
                else:
                    tx.rollback() # Rollback otomatis jika gagal
                    return {"status": "ERROR", "code": response.status_code}
                    
            except Exception as e:
                tx.rollback()
                raise SystemError(f"[CRITICAL] n8n Factory Unreachable: {str(e)}")
