### Verifikasi Status Repository
```bash
#!/bin/bash
# verify-repos.sh

REPOS_DIR="/opt/aios/repos"

echo "Verifying repository status..."

for repo in $REPOS_DIR/*; do
    if [ -d "$repo" ]; then
        cd "$repo"
        if git status > /dev/null 2>&1; then
            echo "✅ $repo - OK"
        else
            echo "❌ $repo - Error"
        fi
    fi
done
```
