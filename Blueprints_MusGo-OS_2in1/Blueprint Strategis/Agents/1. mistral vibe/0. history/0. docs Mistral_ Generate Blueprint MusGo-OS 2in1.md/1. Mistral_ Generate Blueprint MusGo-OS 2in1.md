### **Opsi 2: Gunakan Script untuk Buat Struktur Otomatis**
Jika Anda **malas membuat manual**, buat file `setup_repo.py` dan jalankan:
```python
# setup_repo.py
import os
from pathlib import Path

# Struktur repo yang sudah di-audit
structure = {
    "constitution": [],
    "core": [
        f"Layer_{i:02d}_{name}" for i, name in enumerate([
            "Governor", "Kernel", "Musyawarah_Planner", "GoRo_Executor",
            "Decision_Consensus", "Agent_Orchestration", "Policy_Ethics_Guard",
            "EventBus", "Persistence_Memory", "API_Gateway", "Contract_Registry",
            "RiskEngine", "Billing_Credit", "Security_Network",
            "Observability_Audit", "Adaptive_SelfLearning", "Threat_Intelligence",
        ], 1)
    ],
    "apps/ecosystem/browser": [
        "core/engine", "core/session", "core/ai", "core/config",
        "ui/components", "ui/styles", "ui/themes", "ui/utils",
        "extensions/sandbox", "extensions/api",
        "sdk/python", "sdk/typescript/components", "sdk/typescript/utils",
        "tests/unit", "tests/integration", "tests/e2e", "docs",
    ],
    # Tambahkan struktur lain di sini...
}

def create_structure(base_path: Path, structure: dict):
    for folder, subfolders in structure.items():
        (base_path / folder).mkdir(parents=True, exist_ok=True)
        for subfolder in subfolders:
            (base_path / folder / subfolder).mkdir(parents=True, exist_ok=True)
        # Buat __init__.py
        (base_path / folder / "__init__.py").touch()

if __name__ == "__main__":
    repo_path = Path("Clean_MusGo-OS-2in1_Repo")
    repo_path.mkdir(exist_ok=True)
    create_structure(repo_path, structure)
    print(f"Repo created at: {repo_path.absolute()}")
```

Jalankan:
```bash
python setup_repo.py
```

### **PAR 1: Constitution (Layer 00) - Sovereign Governor**
*(Prioritas pertama, karena ini fondasi untuk semua layer)*

**Konten yang akan di-cover:**
1. **`constitution/__init__.py`**
   - Inisialisasi modul constitution.
   - Import semua komponen (governor, guard, override).

2. **`constitution/GOMU_OS_PRIVATE_KEY.pem`**
   - Private key untuk **digital signing** (contoh struktur, tidak actual key).

3. **`constitution/GOMU_OS_PUBLIC_KEY.pem`**
   - Public key untuk **verifikasi signature**.

4. **`constitution/GOMU_OS_SOVEREIGN_MASTER_CONSTITUTION.sig`**
   - File **constitution** yang sudah **ditandatangani** (signed).
   - Berisi aturan-aturan **immutable** untuk sistem.

5. **`constitution/governor.py`**
   - **Sovereign Governor**: Otak utama yang mengatur **semua layer**.
   - Fitur:
     - **Constitutional Validation**: Validasi semua tindakan terhadap constitution.
     - **Emergency Override**: Mekanisme override untuk keadaan darurat.
     - **Policy Enforcement**: Penegakan kebijakan secara otomatis.
     - **Audit Trail**: Pencatatan semua keputusan.

6. **`constitution/constitutional_guard.py`**
   - **Guard** untuk memastikan **semua operasi** mematuhi constitution.
   - Fitur:
     - **Pre-execution Validation**: Validasi sebelum eksekusi.
     - **Runtime Monitoring**: Pemantauan saat runtime.
     - **Violation Handling**: Penanganan pelanggaran.

7. **`constitution/emergency_override.py`**
   - Mekanisme **override darurat** (misal: shutdown sistem, lockdown, dll.).
   - Fitur:
     - **Manual Override**: Override oleh admin.
     - **Automated Override**: Override otomatis jika terdeteksi ancaman kritis.
     - **Recovery Protocol**: Protokol pemulihan pasca-override.

```text
"""
Sovereign Governor Implementation
=================================
Kelas utama untuk **Sovereign Governor** (Layer 00).
"""

import threading
import hashlib
import json
from typing import Dict, List, Optional, Any, Set, Tuple
from dataclasses import dataclass, field
from enum import Enum, auto
from datetime import datetime
import uuid

from musgo_os_2in1 import (
    ConstitutionalRule,
    GovernanceDecision,
    EmergencyLevel,
    BrowserError,
    SecurityError,
    ValidationError,
    logger,
)

# --- ENUM ---
class DecisionType(Enum):
    """Tipe keputusan governor."""
    POLICY_ENFORCEMENT = auto()    # Penegakan kebijakan
    RESOURCE_ALLOCATION = auto()   # Alokasi resource
    SECURITY_OVERRIDE = auto()     # Override keamanan
    EMERGENCY_ACTION = auto()      # Tindakan darurat
    SYSTEM_SHUTDOWN = auto()       # Shutdown sistem

class GovernanceMode(Enum):
    """Mode governance."""
    STRICT = auto()      # Ketat (semua aturan dijalankan)
    MODERATE = auto()    # Sedang (beberapa aturan bisa di-bypass)
    PERMISSIVE = auto()  # Longgar (hanya aturan kritis yang dijalankan)

# --- DATA STRUCTURE ---
@dataclass
class GovernanceLog:
    """Log keputusan governance."""
    log_id: str = field(default_factory=lambda: str(uuid.uuid4()))
    decision_type: DecisionType
    action: str
    context: Dict[str, Any] = field(default_factory=dict)
    timestamp: datetime = field(default_factory=datetime.now)
    result: str = "SUCCESS"
    metadata: Dict[str, Any] = field(default_factory=dict)

@dataclass
class ConstitutionalViolation:
    """Pelanggaran constitution."""
    violation_id: str = field(default_factory=lambda: str(uuid.uuid4()))
    rule_id: str
    description: str
    severity: str = "CRITICAL"  # LOW, MEDIUM, HIGH, CRITICAL
    timestamp: datetime = field(default_factory=datetime.now)
    context: Dict[str, Any] = field(default_factory=dict)

# --- SOVEREIGN GOVERNOR ---
class SovereignGovernor:
    """
    Governor utama untuk MusGo-OS 2in1.
    Bertanggung jawab untuk:
    - Penegakan constitution
    - Pengambilan keputusan
    - Koordinasi antar layer
    - Emergency override
    """

    def __init__(self, constitution_path: str):
        self.constitution_path = constitution_path
        self._rules: Dict[str, ConstitutionalRule] = {}
        self._logs: List[GovernanceLog] = []
        self._violations: List[ConstitutionalViolation] = []
        self._lock = threading.RLock()
        self._mode = GovernanceMode.STRICT
        self._load_constitution()
        logger.info("SovereignGovernor initialized")

    def _load_constitution(self) -> None:
        """Muat constitution dari file."""
        try:
            with open(self.constitution_path, "r", encoding="utf-8") as f:
                constitution_data = json.load(f)
                for rule_data in constitution_data.get("rules", []):
                    rule = ConstitutionalRule(**rule_data)
                    self._rules[rule.rule_id] = rule
            logger.info(f"Loaded constitution from {self.constitution_path}")
        except Exception as e:
            logger.error(f"Failed to load constitution: {e}")
            raise SecurityError(f"Constitution load failed: {e}")

    def validate_action(
        self,
        action: str,
        context: Dict[str, Any],
        layer: str,
    ) -> Tuple[bool, Optional[ConstitutionalViolation]]:
        """
        Validasi tindakan terhadap constitution.
        Returns:
            Tuple[bool, Optional[ConstitutionalViolation]]: (allowed, violation)
        """
        with self._lock:
            for rule_id, rule in self._rules.items():
                if not rule.enabled:
                    continue

                # Cek apakah action melanggar rule
                if self._check_violation(rule, action, context, layer):
                    violation = ConstitutionalViolation(
                        rule_id=rule.rule_id,
                        description=f"Action '{action}' violates rule '{rule.name}'",
                        severity=rule.severity,
                        context=context,
                    )
                    self._violations.append(violation)
                    logger.warning(
                        f"Constitutional Violation | Rule: {rule.name} | "
                        f"Action: {action} | Layer: {layer}"
                    )
                    return False, violation

            return True, None

    def _check_violation(
        self,
        rule: ConstitutionalRule,
        action: str,
        context: Dict[str, Any],
        layer: str,
    ) -> bool:
        """Cek apakah action melanggar rule."""
        # Cek apakah action match dengan pattern rule
        if rule.pattern and not self._match_pattern(action, rule.pattern):
            return False

        # Cek apakah layer terpengaruh
        if rule.applicable_layers and layer not in rule.applicable_layers:
            return False

        # Cek kondisi tambahan
        if rule.condition:
            try:
                # Evaluasi condition (contoh: context["user_role"] == "admin")
                if not eval(rule.condition, {}, context):
                    return False
            except:
                return False

        return True

    def _match_pattern(self, action: str, pattern: str) -> bool:
        """Cek apakah action match dengan pattern (regex)."""
        import re
        return bool(re.search(pattern, action, re.IGNORECASE))

    def make_decision(
        self,
        decision_type: DecisionType,
        action: str,
        context: Dict[str, Any],
        layer: str,
    ) -> GovernanceDecision:
        """
        Buat keputusan governance.
        Returns:
            GovernanceDecision: Keputusan yang diambil
        """
        with self._lock:
            # Validasi action
            allowed, violation = self.validate_action(action, context, layer)
            if not allowed:
                return GovernanceDecision(
                    decision_id=str(uuid.uuid4()),
                    decision_type=decision_type,
                    action=action,
                    allowed=False,
                    reason=f"Violates rule: {violation.rule_id}",
                    context=context,
                    timestamp=datetime.now(),
                )

            # Catat log
            log = GovernanceLog(
                decision_type=decision_type,
                action=action,
                context=context,
                result="SUCCESS",
            )
            self._logs.append(log)

            logger.info(
                f"Governance Decision | Type: {decision_type.name} | "
                f"Action: {action} | Layer: {layer}"
            )

            return GovernanceDecision(
                decision_id=str(uuid.uuid4()),
                decision_type=decision_type,
                action=action,
                allowed=True,
                reason="Action allowed by constitution",
                context=context,
                timestamp=datetime.now(),
            )

    def set_mode(self, mode: GovernanceMode) -> None:
        """Set mode governance."""
        with self._lock:
            self._mode = mode
            logger.info(f"Governance mode set to: {mode.name}")

    def get_violations(self, limit: int = 100) -> List[ConstitutionalViolation]:
        """Dapatkan daftar pelanggaran."""
        with self._lock:
            return self._violations[-limit:]

    def get_logs(self, limit: int = 100) -> List[GovernanceLog]:
        """Dapatkan daftar log."""
        with self._lock:
            return self._logs[-limit:]

    def cleanup(self) -> None:
        """Bersihkan resource."""
        with self._lock:
            self._rules.clear()
            self._logs.clear()
            self._violations.clear()
            logger.info("SovereignGovernor cleaned up")

Constitutional Guard Implementation

Kelas untuk menegakkan aturan constitution.

from typing import Dict, List, Optional, Any, Callable
from dataclasses import dataclass, field

import uuid

# --- CONSTITUTIONAL GUARD ---
class ConstitutionalGuard:
    """
    Guard untuk memastikan semua operasi mematuhi constitution.
    """

    def __init__(self, governor: "SovereignGovernor"):
        self.governor = governor
        self._hooks: Dict[str, List[Callable]] = {}
        self._lock = threading.RLock()
        logger.info("ConstitutionalGuard initialized")

    def register_hook(
        self,
        event_type: str,
        callback: Callable[[Dict[str, Any]], bool],
    ) -> None:
        """Daftarkan hook untuk event."""
        with self._lock:
            if event_type not in self._hooks:
                self._hooks[event_type] = []
            self._hooks[event_type].append(callback)
            logger.info(f"Registered hook for event: {event_type}")

    def enforce(
        self,
        action: str,
        context: Dict[str, Any],
        layer: str,
    ) -> bool:
        """
        Tegakkan constitution untuk action.
        Returns:
            bool: True jika diizinkan, False jika ditolak
        """
        allowed, violation = self.governor.validate_action(action, context, layer)
        if not allowed:
            logger.warning(
                f"Constitutional Guard Blocked | Action: {action} | "
                f"Violation: {violation.rule_id}"
            )
            return False

        # Panggil hooks
        for event_type, callbacks in self._hooks.items():
            for callback in callbacks:
                if not callback(context):
                    logger.warning(f"Hook blocked action: {action} | Event: {event_type}")
                    return False

        return True

    def cleanup(self) -> None:
        """Bersihkan resource."""
        with self._lock:
            self._hooks.clear()
            logger.info("ConstitutionalGuard cleaned up")

"""
Emergency Override Protocol
===========================
Mekanisme override darurat.
"""

import threading
from typing import Dict, List, Optional, Any
from dataclasses import dataclass, field
from enum import Enum, auto
from datetime import datetime
import uuid

from musgo_os_2in1 import (
    EmergencyLevel,
    EmergencyAction,
    logger,
)

# --- ENUM ---
class OverrideStatus(Enum):
    """Status override."""
    ACTIVE = auto()
    INACTIVE = auto()
    PENDING = auto()

# --- DATA STRUCTURE ---
@dataclass
class EmergencyOverride:
    """Override darurat."""
    override_id: str = field(default_factory=lambda: str(uuid.uuid4()))
    action: EmergencyAction
    level: EmergencyLevel
    status: OverrideStatus = OverrideStatus.PENDING
    triggered_by: str = "system"
    timestamp: datetime = field(default_factory=datetime.now)
    context: Dict[str, Any] = field(default_factory=dict)

# --- EMERGENCY OVERRIDE PROTOCOL ---
class EmergencyOverrideProtocol:
    """
    Protokol untuk emergency override.
    """

    def __init__(self, governor: "SovereignGovernor"):
        self.governor = governor
        self._overrides: List[EmergencyOverride] = []
        self._lock = threading.RLock()
        logger.info("EmergencyOverrideProtocol initialized")

    def trigger_override(
        self,
        action: EmergencyAction,
        level: EmergencyLevel,
        triggered_by: str = "system",
        context: Optional[Dict[str, Any]] = None,
    ) -> EmergencyOverride:
        """Trigger emergency override."""
        with self._lock:
            override = EmergencyOverride(
                action=action,
                level=level,
                triggered_by=triggered_by,
                context=context or {},
            )
            self._overrides.append(override)
            logger.warning(
                f"Emergency Override Triggered | Action: {action.name} | "
                f"Level: {level.name} | By: {triggered_by}"
            )
            return override

    def resolve_override(self, override_id: str) -> bool:
        """Selesaikan override."""
        with self._lock:
            for override in self._overrides:
                if override.override_id == override_id:
                    override.status = OverrideStatus.INACTIVE
                    logger.info(f"Resolved override: {override_id}")
                    return True
            return False

    def get_active_overrides(self) -> List[EmergencyOverride]:
        """Dapatkan override aktif."""
        with self._lock:
            return [o for o in self._overrides if o.status == OverrideStatus.ACTIVE]

    def cleanup(self) -> None:
        """Bersihkan resource."""
        with self._lock:
            self._overrides.clear()
            logger.info("EmergencyOverrideProtocol cleaned up")
```
